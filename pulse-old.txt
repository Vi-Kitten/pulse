/** @template T @typedef {[] | [T]} Maybe
 */

/**
 * @template {unknown[]} Args
 * @template Ret
 * @param {(...xs: Args) => Ret} f 
 * @returns {(...xs: Args) => Ret}
 */
function once(f) {
    /** @type {Maybe<Ret>} */
    let val = [];
    return (...args) => {
        if (val.length) return val[0];
        let ret = f(...args);
        val = [ret];
        return ret;
    };
};

async function painted() {
    await new Promise(res =>
        requestAnimationFrame(() => requestAnimationFrame(res))
    )
};

/**
 * @template T
 * @typedef {{
 *      submit: () => T
 * }} Mounted
 */

/**
 * @template T
 */
class Component {
    /**
     * @param {(render: (elems: HTMLElement[]) => void, flush: () => void) => Mounted<T>} mountFn 
     */
    constructor(mountFn) {
        /** @type {(render: (elems: HTMLElement[]) => void, flush: () => void) => Mounted<T>} */
        this.mount = (render, flush) => {
            let submitted = false;
            const mounted = mountFn(render, flush);
            return {
                submit() {
                    if (submitted) throw new Error("component submitted twice");
                    submitted = true;
                    return mounted.submit();
                }
            };
        };
        /** @type {(render: (elems: HTMLElement[]) => void, resolve: (val: T) => void) => () => void} */
        this.root = (render, resolve) => {
            /** @type {Maybe<Mounted<T>>} */
            let mounted = [];
            const run = once(() => resolve(mounted.length
                ? mounted[0].submit()
                : (() => { throw new Error("Internal rooting state error")})()
            ));
            let fastFlush = false;
            mounted = [mountFn(render, () => mounted.length
                ? run()
                : (() => { fastFlush = true; })()
            )];
            if (fastFlush) run();
            return () => { run() };
        };
    }
    /**
     * Extract value from component without rendering.
     * @returns {T}
     */
    rhetorical() {
        return this.mount(_ => {}, () => {}).submit()
    }
    /**
     * Modify return without effecting behaviour.
     * @template U
     * @param {(x: T) => U} f 
     * @returns {Component<U>}
     */
    map(f) {
        return new Component((render, resolve) => {
            let mounted = this.mount(render, resolve);
            return {
                submit: () => f(mounted.submit())
            };
        });
    }
    /**
     * Assert that a component will never be submitted.
     * @param {(val: T) => void} f 
     * @returns {Component<never>}
     */
    diverge(f = x => {}) {
        return new Component((render, _) => {
            this.root(render, f);
            return {
                submit: () => {
                    throw new Error("Divergent component submitted")
                }
            };
        });
    }
    /**
     * Render with a continuation after submition.
     * 
     * _Sequential operator_
     * @template U
     * @template [Ret=never]
     * @param {(val: T, early: (ret: Ret) => void) => Component<U>} f
     * @returns {Component<U | Ret>}
     */
    then(f) {
        return new Component((render, flush) => {
            /** @type {Maybe<Mounted<U | Ret>>} */
            let mountedInner = [];
            let run = this.root(render, val =>
                mountedInner = [Component.do(ret => f(val, ret)).mount(render, flush)]
            );
            return {
                submit() {
                    run();
                    if (mountedInner.length == 0) throw new Error("Internal sequencing error");
                    return mountedInner[0].submit();
                }
            };
        });
    }
    /**
     * Apply component of function to components of arguments, rendered together.
     * 
     * _Parallel operator_
     * @template {unknown[]} Args
     * @template Ret
     * @this {Component<(...args: Args) => Ret>}
     * @param  {{[K in keyof Args]: Component<Args[K]>}} compArgs
     * @returns {Component<Ret>}
     */
    applyTo(...compArgs) {
        let arrComp = /** @type {Component<[(...xs: Args) => Ret, ...Args]>} */ (Component.array([this, ...compArgs]));
        return arrComp.map(([f, ...args]) => f(...args));
    }
    /**
     * Provide early return to a component.
     * @template T
     * @template Ret
     * @param {(early: (ret: Ret) => void) => Component<T>} f
     * @returns {Component<T | Ret>}
     */
    static do(f) {
        return new Component((render, flush) => {
            /** @type {Maybe<Ret>} */
            let ret = [];
            const inner = f(once(val => {
                ret = [val];
                flush();
            }));
            const mounted = inner.mount(render, flush);
            return {
                submit: () => ret.length ? ret[0] : mounted.submit()
            };
        });
    }
    /**
     * Short circuit rendering and produce a value.
     * 
     * _Series unit_
     * @template T
     * @param {T} val
     * @returns {Component<T>}
     */
    static done(val) {
        return new Component((_, flush) => {
            flush();
            return {
                submit: () => val
            };
        })
    }
    /**
     * Repeatedly render and change value.
     * @template T
     * @param {T} init
     * @param {(state: T, cont: (val: T) => void, brk: (ret: T) => void) => Component<T>} f
     * @returns {Component<T>}
     */
    static loop(init, f) {
        return Component.loopWith(init, f, x => x);
    }
    /**
     * Repeatedly render and change value, breaking or projecting to a different type.
     * @template T
     * @template U
     * @param {T} init
     * @param {(state: T, cont: (val: T) => void, brk: (ret: U) => void) => Component<T>} f
     * @param {(state: T) => U} proj
     * @returns {Component<U>}
     */
    static loopWith(init, f, proj) {
        return new Component((render, flush) => {
            let state = init;
            /** @type {Maybe<U>} */
            let final = [];
            function step() {
                // trampolined recursion
                let looping = true, immediate = true;
                function stepInner() {
                    if (looping) {
                        immediate = true;
                    } else {
                        step();
                    }
                }
                
                while (immediate) {
                    if (final.length != 0) break;
                    immediate = false;

                    Component.do(cont => f(state, cont, val => {
                        // on break
                        final = [val];
                        flush();
                    })).root(render, val => {
                        // on continue
                        state = val;
                        stepInner();
                    });
                }

                looping = false;
            }
            step();
            return {
                submit() {
                    if (final.length == 0) {
                        final = [proj(state)]
                    }
                    return final[0];
                }
            };
        });
    }
    /**
     * Wrap the value in a component without rendering effects.
     * 
     * _Parallel unit_
     * @template T
     * @param {T} val
     * @returns {Component<T>}
     */
    static wrap(val) {
        return new Component((render, _) => {
            render([]);
            return {
                submit: () => val
            };
        })
    }
    /**
     * Turn an array of components into a component for an array of values, rendered together.
     * 
     * _Parallel operator_
     * @template {unknown[]} Args
     * @param  {{[K in keyof Args]: Component<Args[K]>}} components
     * @returns {Component<Args>}
     */
    static array(components) {
        return new Component((render, flush) => {
            const renderBuffer = Array(components.length).fill(null);
            const mounts = components.map((component, i) => component.mount(
                elems => {
                    renderBuffer[i] = elems;
                    if (renderBuffer.includes(null)) return;
                    render(renderBuffer.flat());
                },
                flush
            ));
            return {
                submit: () => /** @type {Args} */ (mounts.map(mount => mount.submit()))
            }
        });
    }
}

        super("value", { bubbles: true, cancelable: true, composed: true });
    }
}

function Html(html, spec = {}) {
    const temp = document.createElement('template');
    temp.innerHTML = html.trim();
    const elems = [...temp.content.children];
    let handle = undefined;
    if (spec.style !== undefined) {
        for (const elem of elems) {
            elem.style = spec.style;
        }
    }
    if (spec.onClick !== undefined) {
        for (const elem of elems) {
            elem.onclick = async (e) => handle(spec.onClick(e))
        }
    }
    let val = null;
    return Component((render, flush) => {
        handle = eventVal => {
            val = eventVal;
            for (const elem of elems) {
                elem.inert = true;
            }
            flush();
        };
        render(elems);
        return () => val;
    })
}

function Tag(tagName, inner, spec = {}) {
    const elem = document.createElement(tagName);
    if (spec.style !== undefined) {
        elem.style = spec.style;
    }
    return Component((render, flush) => {
        render([elem]);
        return inner.mount(children => elem.replaceChildren(...children), flush);
    });
}

async function renderRoot(comp) {
    let val = await new Promise(res => comp.root(
        elems => document.body.replaceChildren(...elems),
        res
    ));
    document.body.replaceChildren();
    if (val !== undefined) {
        await painted();
        alert(val);
    }
    location.replace("about:blank");
}
